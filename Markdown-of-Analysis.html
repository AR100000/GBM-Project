Geometric Brownian Motion Equation

\[ S_t = S_0 \exp\left(\left(\mu - \frac{\sigma^2}{2}\right)t + \sigma W_t\right) \]

Estimate Annual Drift and Velocity

\[ \hat{\sigma}=s_r\sqrt{252} \]

where the daily log-return standard deviation is

\[ s_r=\sqrt{\frac{1}{n-1}\sum_{t=1}^{n}(r_t-\bar r)^2} \]

  • \(S_t\) is the stock price at time \(t\).
  • \(r_t\) is the daily log return.
  • \(\bar r=\frac{1}{n}\sum_{t=1}^{n}r_t\) is the mean daily log return.
  • \(n\) is the number of daily log returns.
  • \(252\) is the number of trading days per year.

Annualised Drift

\[ \hat{\mu}=252\bar r+\frac{1}{2}\hat{\sigma}^{\,2} \]

Substituting the volatility estimate gives

\[ \hat{\mu}=252\bar r+\frac{1}{2}(252s_r^2) \]

Equivalently,

\[ \hat{\mu}=252\left(\bar r+\frac{1}{2}s_r^2\right) \]

where \(\hat{\mu}\) is the estimated annualised GBM drift parameter.

The additional term \(\frac12\hat{\sigma}^2\) arises from the GBM model:

It adjusts the annualised mean log return to estimate the drift parameter \(\mu\).

Autocorrelation of Squared Returns

Checks if there are periods of consistent high returns or low returns - essentially checking if there is a correlation between the absolute value of the returns across different time periods. Spikes in the ACF that extend past the confidence band (dashed blue line) show that there is some correlation.

compare_window <- function(prices, window, forecast_days = 21,
                           trading_days = 252) {
  
  # Error Handling
  if (length(prices) < window + forecast_days) {
    return(data.frame(
      Mean_APE = NA_real_,
      Median_APE = NA_real_,
      Mean_Annual_Volatility = NA_real_,
      Forecasts = 0
    ))
  }
  
  errors <- numeric(0)
  volatility_estimates <- numeric(0)
  
  # Rolling forecasts
  starts <- seq(
    from = window + 1,
    to = length(prices) - forecast_days + 1,
    by = forecast_days
  )
  
  for (i in starts) {
    
    estimation_prices <- prices[(i - window):(i - 1)]
    returns <- diff(log(estimation_prices))
    
    sigma_hat <- sd(returns) * sqrt(trading_days)
    mu_hat <- mean(returns) * trading_days +
      0.5 * sigma_hat^2
    
    S0 <- prices[i - 1]
    
    # Expected price after the forecast period
    predicted_price <- S0 * exp(
      mu_hat * forecast_days / trading_days
    )
    
    actual_price <- prices[i + forecast_days - 1]
    
    error <- abs(predicted_price - actual_price) /
      actual_price * 100
    
    errors <- c(errors, error)
    volatility_estimates <- c(volatility_estimates, sigma_hat)
  }
  
  data.frame(
    Mean_APE = mean(errors),
    Median_APE = median(errors),
    Mean_Annual_Volatility = mean(volatility_estimates),
    Forecasts = length(errors)
  )
}

# Compare each estimation window
comparison_list <- lapply(windows, function(w) {
  compare_window(prices, w, forecast_days, trading_days)
})

# stock-level gbm statistics

industry_stock_results <- list()

for (industry in names(industry_tickers)) {
  
  tickers <- intersect(
    industry_tickers[[industry]],
    names(data_list)
  )
  
  for (ticker in tickers) {
    
    stock <- data_list[[ticker]]
    
    date_col <- which(tolower(names(stock)) == "date")
    close_col <- which(tolower(names(stock)) == "close")
    
    if (length(date_col) != 1 || length(close_col) != 1) {
      next
    }
    
    dates <- as.Date(stock[[date_col]])
    prices <- suppressWarnings(
      as.numeric(as.character(stock[[close_col]]))
    )
    
    # Use Existing Parameters
    keep <- !is.na(dates) &
      dates >= start_date &
      dates <= end_date &
      is.finite(prices) &
      prices > 0
    
    dates <- dates[keep]
    prices <- prices[keep]
    
    # Sort prices chronologically
    ordering <- order(dates)
    prices <- prices[ordering]
    
    if (length(prices) < 30) {
      next
    }
    
    # Calculate daily log returns
    log_returns <- diff(log(prices))
    log_returns <- log_returns[is.finite(log_returns)]
    
    if (length(log_returns) < 20 ||
        !is.finite(sd(log_returns)) ||
        sd(log_returns) == 0) {
      next
    }
    
    # Estimate GBM parameters using existing trading_days
    sigma_hat <- sd(log_returns) * sqrt(trading_days)
    
    mu_hat <- mean(log_returns) * trading_days +
      0.5 * sigma_hat^2
    
    # Return distribution statistics
    standardised_returns <- (
      log_returns - mean(log_returns)
    ) / sd(log_returns)
    
    industry_stock_results[[length(industry_stock_results) + 1]] <-
      data.frame(
        Industry = industry,
        Ticker = ticker,
        Observations = length(log_returns),
        Mean_Daily_Log_Return = mean(log_returns),
        Annualised_Volatility = sigma_hat,
        Annualised_GBM_Drift = mu_hat,
        Skewness = mean(standardised_returns^3),
        Excess_Kurtosis = mean(standardised_returns^4) - 3
      )
  }
}

  # Check industry assignment
  if (!ticker %in% assigned_tickers) {
    return(data.frame(
      Ticker = ticker,
      Reason = "Not assigned to an industry"
    ))
  }

  stock <- data_list[[ticker]]

  date_col <- which(tolower(names(stock)) == "date")
  close_col <- which(tolower(names(stock)) == "close")

  if (length(date_col) != 1 || length(close_col) != 1) {
    return(data.frame(
      Ticker = ticker,
      Reason = "Missing or invalid date/close column"
    ))
  }

  dates <- tryCatch(
    as.Date(stock[[date_col]]),
    error = function(e) rep(as.Date(NA), nrow(stock))
  )

  prices <- suppressWarnings(
    as.numeric(as.character(stock[[close_col]]))
  )

  keep <- !is.na(dates) &
    dates >= start_date &
    dates <= end_date &
    is.finite(prices) &
    prices > 0

  dates <- dates[keep]
  prices <- prices[keep]

  prices <- prices[order(dates)]

  if (length(prices) < 30) {
    return(data.frame(
      Ticker = ticker,
      Reason = "Fewer than 30 valid prices"
    ))
  }

  log_returns <- diff(log(prices))
  log_returns <- log_returns[is.finite(log_returns)]

  if (length(log_returns) < 20) {
    return(data.frame(
      Ticker = ticker,
      Reason = "Fewer than 20 valid returns"
    ))
  }

  if (!is.finite(sd(log_returns)) || sd(log_returns) == 0) {
    return(data.frame(
      Ticker = ticker,
      Reason = "Zero or invalid return volatility"
    ))
  }

  data.frame(Ticker = ticker, Reason = "Analysed")
})

#Actual GBM Simulation per Industry

set.seed(123)

n_simulations <- 500
trading_days <- 252

gbm_fit_results <- list()

for (i in seq_len(nrow(industry_stock_results))) {

  ticker <- industry_stock_results$Ticker[i]
  industry <- industry_stock_results$Industry[i]

  stock <- data_list[[ticker]]

  if (is.null(stock)) next

  # Identify required columns
  date_col <- which(tolower(names(stock)) == "date")
  close_col <- which(tolower(names(stock)) == "close")

  if (length(date_col) != 1 || length(close_col) != 1) next

  dates <- as.Date(stock[[date_col]])
  prices <- suppressWarnings(
    as.numeric(as.character(stock[[close_col]]))
  )

  # Apply selected date range and price validity checks
  keep <- !is.na(dates) &
    dates >= start_date &
    dates <= end_date &
    is.finite(prices) &
    prices > 0

  dates <- dates[keep]
  prices <- prices[keep]

  # Sort and remove duplicate dates
  ordering <- order(dates)
  dates <- dates[ordering]
  prices <- prices[ordering]

  unique_dates <- !duplicated(dates)
  dates <- dates[unique_dates]
  prices <- prices[unique_dates]

  if (length(prices) < 30) next

  # Calculate log returns
  log_returns <- diff(log(prices))

  if (length(log_returns) < 20 ||
      !is.finite(sd(log_returns)) ||
      sd(log_returns) == 0) next

  # Estimate annualised GBM parameters
  sigma_hat <- sd(log_returns) * sqrt(trading_days)

  mu_hat <- mean(log_returns) * trading_days +
    0.5 * sigma_hat^2

  # Number of intervals between observed prices
  n_days <- length(prices) - 1

  # Simulate daily log-price increments
  simulated_increments <- matrix(
    rnorm(
      n_days * n_simulations,
      mean = (mu_hat - 0.5 * sigma_hat^2) / trading_days,
      sd = sigma_hat / sqrt(trading_days)
    ),
    nrow = n_days,
    ncol = n_simulations
  )

  # Construct simulated price paths
  simulated_paths <- rbind(
    rep(prices[1], n_simulations),
    prices[1] * exp(
      apply(simulated_increments, 2, cumsum)
    )
  )

  # Pointwise 95% simulation interval and median path
  lower_95 <- apply(
    simulated_paths, 1, quantile, probs = 0.025
  )

  median_path <- apply(
    simulated_paths, 1, median
  )

  upper_95 <- apply(
    simulated_paths, 1, quantile, probs = 0.975
  )

  # Compare actual prices with simulated distribution
  coverage_95 <- mean(
    prices >= lower_95 & prices <= upper_95
  ) * 100

  median_path_mape <- mean(
    abs(prices - median_path) / prices
  ) * 100

  log_price_rmse <- sqrt(
    mean((log(prices) - log(median_path))^2)
  )

  # Save stock-level results
  gbm_fit_results[[length(gbm_fit_results) + 1]] <-
    data.frame(
      Ticker = ticker,
      Industry = industry,
      Number_of_Prices = length(prices),
      Median_Path_MAPE = median_path_mape,
      Log_Price_RMSE = log_price_rmse,
      Coverage_95_Percent = coverage_95
    )
}

# Error Catcher
# Stop if no stocks could be compared
if (length(gbm_fit_results) == 0) {
  stop("No stocks were successfully simulated.")
}

# Combine results
gbm_fit_stock_results <- do.call(
  rbind, gbm_fit_results
)
